<script>
    google.charts.load('current', {packages:['corechart','bar']});
    google.charts.setOnLoadCallback(drawMultSeries);

    function drawMultSeries() {
        var data = google.visualization.arrayToDataTable([
            ['Country', 'Happiness Score', { role: 'style' }],
            ['Denmark', 7.526, 'green'],
            ['Switzerland', 7.509, 'green'],
            ['Iceland', 7.501, 'green'],
            ['Norway', 7.498, 'green'],
            ['Finland', 7.413, 'green'],
            ['Thailand', 6.474, 'green'],
            ['Benin', 3.484, 'red'],
            ['Afghanistan', 3.36, 'red'],
            ['Togo', 3.303, 'red'],
            ['Syria', 3.069, 'red'],
            ['Burundi', 2.905, 'red'],
        ]);

        var options = {
            title: 'World Happiness Score',
            hAxis: {
                title: 'Country',
                viewWindow: {
                    min: [0, 0, 0],
                    max: [10, 10, 10]
                }
            },
            vAxis: {
                title: 'Happiness Score'
            },
            legend: 'none'  // Hide legend because we're using colors in bars
        };

        var chart = new google.visualization.ColumnChart(document.getElementById('chart_div'));
        chart.draw(data, options);
    }
</script>
